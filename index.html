const express = require('express'); const session = require('express-session'); const multer = require('multer'); const fs = require('fs'); const path = require('path'); const ADMIN_PASSWORD = 'यहाँ-अपना-मज़बूत-पासवर्ड-लिखें'; const SESSION_SECRET = 'यहाँ-कोई-बहुत-लंबा-गुप्त-वाक्य-लिखें'; const DATA_FILE = path.join(__dirname, 'data.json'); const app = express(); app.use(express.json()); app.use(session({ secret: SESSION_SECRET, resave: false, saveUninitialized: false, cookie: { httpOnly: true, maxAge: 1000 * 60 * 60 * 8 } })); function readData() { if (!fs.existsSync(DATA_FILE)) { return { profilePicture: '', posts: [] }; } return JSON.parse(fs.readFileSync(DATA_FILE, 'utf8')); } function writeData(data) { fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2)); } function requireAdmin(req, res, next) { if (req.session.isAdmin) return next(); res.status(401).json({ error: 'सिर्फ़ एडमिन के लिए' }); } const storage = multer.diskStorage({ destination: path.join(__dirname, 'uploads'), filename: (req, file, cb) => { const ext = path.extname(file.originalname).toLowerCase(); cb(null, Date.now() + ext); } }); const upload = multer({ storage, limits: { fileSize: 5 * 1024 * 1024 }, fileFilter: (req, file, cb) => { const ok = ['.jpg', '.jpeg', '.png', '.webp']; cb(null, ok.includes(path.extname(file.originalname).toLowerCase())); } }); app.use('/uploads', express.static(path.join(__dirname, 'uploads'))); app.use(express.static(path.join(__dirname, 'public'), { index: 'index.html' })); app.post('/api/login', (req, res) => { if (req.body.password === ADMIN_PASSWORD) { req.session.isAdmin = true; return res.json({ ok: true }); } res.status(401).json({ error: 'पासवर्ड ग़लत है' }); }); app.post('/api/logout', (req, res) => { req.session.destroy(() => res.json({ ok: true })); }); app.get('/api/me', (req, res) => { res.json({ isAdmin: !!req.session.isAdmin }); }); app.get('/api/data', (req, res) => { res.json(readData()); }); app.post('/api/profile-picture', requireAdmin, upload.single('image'), (req, res) => { if (!req.file) return res.status(400).json({ error: 'तस्वीर नहीं मिली' }); const data = readData(); data.profilePicture = '/uploads/' + req.file.filename; writeData(data); res.json({ ok: true }); }); app.post('/api/posts', requireAdmin, upload.single('image'), (req, res) => { const data = readData(); data.posts.unshift({ id: Date.now(), category: req.body.category || 'General', title: req.body.title || '', text: req.body.text || '', image: req.file ? '/uploads/' + req.file.filename : '', date: new Date().toISOString() }); writeData(data); res.json({ ok: true }); }); app.delete('/api/posts/:id', requireAdmin, (req, res) => { const data = readData(); data.posts = data.posts.filter(p => String(p.id) !== req.params.id); writeData(data); res.json({ ok: true }); }); app.get('/admin', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin.html')); }); app.listen(3000, () => console.log('सर्वर चालू: http://localhost:3000'));
